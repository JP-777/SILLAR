#!/usr/bin/env bash
# Runner focal de [M06-CICLO], v2 (24/10/2026). Sustituye al heredoc del 23/10/2026.
#
# Defecto de la v1: se lanzó como `bash <<'BASH' … BASH`, con el script entrando por stdin.
# El arnés e2e lanza sus hijos con stdio 'inherit' (e2e/setup/shell.ts:42-46), y entre ellos está
# `docker compose exec -T db psql -f …` (e2e/setup/docker.ts:167-168, :205-209). `-T` quita el TTY,
# pero no el stdin. Así que un hijo podía leer, y descartar, el resto del script.
# Bash llegaba a EOF justo después del pipeline `| tee`: no escribía M06_CICLO_RC, ni git-status,
# ni SHA256SUMS. Y salía con 0, también cuando la prueba fallaba.
#
# Cambios de la v2:
#   1. Se ejecuta desde un archivo (`bash <este archivo>`), nunca por stdin. Si llega por stdin, se niega.
#   2. Playwright corre con `</dev/null`: ningún hijo hereda un stdin que se pueda consumir.
#   3. El rc se escribe en un archivo propio (m06-ciclo.rc) en cuanto existe, antes de cualquier otro paso.
#   4. La última línea es RUNNER_FIN: si no aparece, el runner no terminó.
#   5. El directorio de evidencias tiene que ser nuevo: no sobrescribe nada histórico.
#   6. Sale con el rc de la prueba, no con el de la última orden.
#
# Corrección del 24/10/2026, ronda 4:
#   7. El preflight rechaza la worktree si tiene cambios rastreados o archivos sin seguimiento no
#      ignorados. Lo comprueba antes de crear $EV y antes de cualquier e2e.
#   8. Cada evidencia obligatoria se comprueba: m06-ciclo.log, m06-ciclo.rc, la línea
#      M06_CICLO_RC, git-status.txt, un playwright-report de esta corrida y SHA256SUMS.txt
#      verificado. Si alguna falla, sale con 70 y M06_FOCAL=EVIDENCIA_INVALIDA, sin RUNNER_FIN.
#      El rc real de Playwright solo se devuelve si todas están completas.
#
# Corrección del 24/10/2026, cierre:
#   9. El estado final queda dentro de $EV, no solo en la terminal. m06-final.txt guarda
#      M06_CICLO_RC, M06_FOCAL, RUNNER_EXIT, SHA256SUMS_SHA256 y, como última línea,
#      RUNNER_FIN=1. Se sella con m06-final.sha256. Se escribe solo cuando SHA256SUMS.txt
#      ya verifica, y se vuelve a comprobar antes de salir.
#      Si algo falla, m06-final.txt dice M06_FOCAL=EVIDENCIA_INVALIDA y no lleva RUNNER_FIN.
#      Para comprobarlo:  cd $EV && sha256sum -c m06-final.sha256 && sha256sum -c SHA256SUMS.txt
#      y que SHA256SUMS_SHA256 sea el sha256 de SHA256SUMS.txt.
#
# Uso:  bash QA-M06-3068A55-RUNNER-FOCAL-V2-20261024.sh.txt  [WT] [EV]
set -u

if [[ ! -f "${BASH_SOURCE[0]:-}" ]]; then
  echo "STOP: este runner se ejecuta desde un archivo, no por stdin ni heredoc" >&2
  exit 2
fi

WT="${1:-$HOME/SILLAR_M06_QA_DF2C947}"
EV="${2:-$HOME/SILLAR_M06_EVIDENCIAS_$(date +%Y%m%d_%H%M%S)}"
BASE="df2c94798a15c9c3354c90c0be86620c13599ed8"  # candidato M05b+D1, autorizado por Chat 3 (FASE B)

if [[ -e "$EV" ]]; then
  echo "STOP: $EV ya existe; no se sobrescriben evidencias" >&2
  exit 2
fi
cd "$WT" || exit 1

echo "===== PREFLIGHT M06 ====="

if [[ "$(git rev-parse HEAD)" != "$BASE" ]]; then
  echo "STOP: HEAD incorrecto"
  exit 1
fi

git diff --check || exit 1

if [[ ! -f e2e/tests/zz-z-m06-ciclo.spec.ts ]]; then
  echo "STOP: Falta la prueba M06"
  exit 1
fi

if ! SUCIO="$(git status --porcelain --untracked-files=normal)"; then
  echo "STOP: no se pudo leer git status"
  exit 1
fi
if [[ -n "$SUCIO" ]]; then
  echo "STOP: worktree con cambios o archivos sin seguimiento; el ciclo solo corre sobre un árbol limpio"
  printf '%s\n' "$SUCIO"
  exit 1
fi

mkdir -p -- "$(dirname -- "$EV")" && mkdir -- "$EV" || { echo "STOP: no se pudo crear $EV" >&2; exit 2; }
date --iso-8601=seconds > "$EV/m06-inicio.txt" || { echo "STOP: no se pudo escribir m06-inicio.txt" >&2; exit 2; }

echo "===== M06 CICLO FÍSICO ====="

(
  cd "$WT/e2e" || exit 1
  env -u E2E_KEEP_STACK pnpm exec playwright test \
    tests/zz-z-m06-ciclo.spec.ts \
    --project=chromium \
    --workers=1 \
    --retries=0
) </dev/null 2>&1 | tee "$EV/m06-ciclo.log"

ESTADOS=("${PIPESTATUS[@]}")
RC=${ESTADOS[0]}
TEE_RC=${ESTADOS[1]:-1}

falla_evidencia() {
  echo "EVIDENCIAS_INCOMPLETAS: $1" >&2
  # El fallo también queda en el expediente. Si ni eso se puede escribir, la salida 70 basta.
  rm -f -- "$EV/m06-final.sha256" "$EV/m06-final.tmp"
  printf 'M06_CICLO_RC=%s\nM06_FOCAL=EVIDENCIA_INVALIDA\nRUNNER_EXIT=70\nMOTIVO=%s\n' "$RC" "$1" \
    > "$EV/m06-final.txt" 2>/dev/null || true
  echo "RC_PLAYWRIGHT=$RC (no vale como resultado sin evidencias completas)" >&2
  echo "EVIDENCIAS=$EV"
  echo "M06_FOCAL=EVIDENCIA_INVALIDA"
  exit 70
}

[[ "$TEE_RC" -eq 0 ]] || falla_evidencia "tee no pudo escribir m06-ciclo.log (rc=$TEE_RC)"
echo "$RC" > "$EV/m06-ciclo.rc" || falla_evidencia "no se pudo escribir m06-ciclo.rc"
echo "M06_CICLO_RC=$RC" >> "$EV/m06-ciclo.log" || falla_evidencia "no se pudo anotar M06_CICLO_RC en m06-ciclo.log"
echo "M06_CICLO_RC=$RC"
[[ "$(cat "$EV/m06-ciclo.rc")" == "$RC" ]] || falla_evidencia "m06-ciclo.rc no contiene el rc"
[[ "$(tail -n 1 "$EV/m06-ciclo.log")" == "M06_CICLO_RC=$RC" ]] || falla_evidencia "m06-ciclo.log no termina en M06_CICLO_RC"

git status --short > "$EV/git-status.txt" || falla_evidencia "no se pudo escribir git-status.txt"

# El reporte está en .gitignore: el preflight no lo ve, y uno viejo pasaría por bueno.
INFORME="$WT/e2e/playwright-report"
[[ -f "$INFORME/index.html" ]] || falla_evidencia "Playwright no dejó playwright-report/index.html"
[[ -n "$(find "$INFORME/index.html" -newer "$EV/m06-inicio.txt")" ]] || falla_evidencia "playwright-report es anterior a esta corrida"
cp -a "$INFORME" "$EV/" || falla_evidencia "no se pudo copiar playwright-report"

(
  set -o pipefail
  cd "$EV" &&
  find . -type f ! -name SHA256SUMS.txt ! -name 'm06-final.*' \
    -print0 | sort -z | xargs -0 -r sha256sum \
    > SHA256SUMS.txt
) || falla_evidencia "no se pudo generar SHA256SUMS.txt"

for f in m06-inicio.txt m06-ciclo.log m06-ciclo.rc git-status.txt playwright-report/index.html; do
  awk -v f="./$f" '$2 == f { ok = 1 } END { exit !ok }' "$EV/SHA256SUMS.txt" \
    || falla_evidencia "SHA256SUMS.txt no incluye $f"
done
( cd "$EV" && sha256sum --check --quiet --strict SHA256SUMS.txt ) || falla_evidencia "SHA256SUMS.txt no verifica"

if [[ "$RC" -eq 0 ]]; then FOCAL=PASS; else FOCAL=REVISAR; fi
SUMS_SHA="$(sha256sum "$EV/SHA256SUMS.txt" | cut -d' ' -f1)" && [[ -n "$SUMS_SHA" ]] \
  || falla_evidencia "no se pudo calcular el sha256 de SHA256SUMS.txt"

printf 'M06_CICLO_RC=%s\nM06_FOCAL=%s\nRUNNER_EXIT=%s\nSHA256SUMS_SHA256=%s\nRUNNER_FIN=1\n' \
  "$RC" "$FOCAL" "$RC" "$SUMS_SHA" > "$EV/m06-final.tmp" \
  && mv -- "$EV/m06-final.tmp" "$EV/m06-final.txt" \
  || falla_evidencia "no se pudo escribir m06-final.txt"
( cd "$EV" && sha256sum m06-final.txt > m06-final.sha256 ) || falla_evidencia "no se pudo sellar m06-final.txt"

# Comprobación de lo que quedó en disco, no de lo que se quiso escribir.
( cd "$EV" && sha256sum --check --quiet --strict m06-final.sha256 \
    && sha256sum --check --quiet --strict SHA256SUMS.txt ) \
  || falla_evidencia "m06-final.txt o SHA256SUMS.txt no verifican tras el cierre"
grep -qx "M06_CICLO_RC=$RC" "$EV/m06-final.txt" \
  && grep -qx "M06_FOCAL=$FOCAL" "$EV/m06-final.txt" \
  && grep -qx "RUNNER_EXIT=$RC" "$EV/m06-final.txt" \
  && grep -qx "SHA256SUMS_SHA256=$(sha256sum "$EV/SHA256SUMS.txt" | cut -d' ' -f1)" "$EV/m06-final.txt" \
  && [[ "$(tail -n 1 "$EV/m06-final.txt")" == "RUNNER_FIN=1" ]] \
  || falla_evidencia "m06-final.txt no corresponde al resultado"

echo "EVIDENCIAS=$EV"
cat "$EV/m06-final.txt"
exit "$RC"
